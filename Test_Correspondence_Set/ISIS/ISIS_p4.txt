#include <core.p4>

// Define Header Fields
header ethernet_t {
    bit<48> dstAddr;
    bit<48> srcAddr;
    bit<16> etherType;
}

header isis_t {
    bit<8> protocol_id;       // Protocol ID
    bit<8> version;           // IS-IS Version
    bit<8> type;              // IS-IS PDU Type
    bit<8> reserved;          // Reserved
    bit<16> checksum;         // Checksum
    bit<32> system_id;        // System ID
}

// Metadata
struct metadata_t {}

// Header Stack
struct headers {
    ethernet_t ethernet;
    isis_t isis;
}

// Parser
parser my_parser(packet_in packet,
                 out headers hdr,
                 inout metadata_t meta,
                 inout standard_metadata_t standard_metadata) {
    state start {
        packet.extract(hdr.ethernet);
        transition select(hdr.ethernet.etherType) {
            0x88cc: parse_isis;  // Ethernet type for IS-IS
            default: accept;
        }
    }

    state parse_isis {
        packet.extract(hdr.isis);
        transition accept;
    }
}

// IS-IS Packet Processing
table isis_packet_processing {
    key = {
        hdr.isis.type: exact;
        hdr.isis.system_id: exact;
    }
    actions = {
        forward_isis;
        drop;
    }
    size = 1024;
    default_action = drop();
}

// Actions
action forward_isis(bit<9> egress_port) {
    standard_metadata.egress_spec = egress_port;
}

action drop() {
    mark_to_drop();
}

// Ingress Pipeline
control ingress {
    apply {
        if (hdr.isis.isValid()) {
            isis_packet_processing.apply();
        } else {
            drop();
        }
    }
}

// Deparser
control deparser(packet_out packet, in headers hdr) {
    apply {
        packet.emit(hdr.ethernet);
        packet.emit(hdr.isis);
    }
}

// Main Program
V1Switch(
    my_parser(),
    ingress(),
    egress(),
    deparser()
) main;
